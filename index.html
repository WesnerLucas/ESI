Código actualizado para Google Apps Script (Code.gs)

function doGet(e) {
  try {
    var props = PropertiesService.getScriptProperties();
    var ronda = parseInt(props.getProperty('RONDA_VOTACION') || '1', 10);

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var values = sheet.getDataRange().getValues();

    var nombresMap = {};
    var votos = [];

    for (var i = 1; i < values.length; i++) {
      var row = values[i];
      var fecha = row[0];
      var sessionId = row[1];
      var nombre = String(row[2] || '').trim();
      var curso = String(row[3] || 'General').trim();
      var categoria = String(row[5] || '').trim();
      var texto = String(row[6] || '').trim();
      var eleccion = String(row[7] || '').trim();

      if (categoria === 'Actividad 2 - Votación') {
        votos.push({
          timestamp: fecha ? new Date(fecha).toISOString() : '',
          sessionId: sessionId,
          curso: curso,
          ronda: texto,
          target: eleccion
        });
      } else if (nombre && nombre.toLowerCase() !== 'anónimo' && nombre.toLowerCase() !== 'anonimo') {
        nombresMap[nombre] = true;
      }
    }

    var nombres = Object.keys(nombresMap).sort(function(a, b) {
      return a.localeCompare(b, 'es');
    });

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ok', ronda: ronda, nombres: nombres, votos: votos }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var props = PropertiesService.getScriptProperties();

    if (data.action === 'nueva_ronda') {
      var currentRound = parseInt(props.getProperty('RONDA_VOTACION') || '1', 10);
      var nextRound = currentRound + 1;
      props.setProperty('RONDA_VOTACION', String(nextRound));
      return ContentService
        .createTextOutput(JSON.stringify({ status: 'ok', ronda: nextRound }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Fecha y Hora', 'ID Sesion', 'Nombre o Apodo', 'Curso',
        'Nro Situacion', 'Categoria', 'Situacion', 'Respuesta', 'Comentario'
      ]);
    }

    var timestamp = new Date();
    var rows = data.respuestas.map(function(item) {
      return [
        timestamp, data.sessionId, data.nombre || 'Anónimo', data.curso || 'General',
        item.id, item.categoria, item.texto, item.eleccion, item.comentario || ''
      ];
    });

    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ok', filas: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
